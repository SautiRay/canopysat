Python:

from canopysat import Forest

# Simple — no API key needed (10 free analyses/month)
result = Forest.analyze(lat=-3.4653, lng=-62.2159, size=10)
print(result.score)           # 80
print(result.canopy_height)   # 13.2m (NASA GEDI LiDAR)
print(result.summary())

# With API key
import requests

API_KEY = "YOUR_API_KEY"
BASE_URL = "https://www.canopysat.org/api/v1"

response = requests.post(f"{BASE_URL}/analyze",
    headers={"Authorization": f"Bearer {API_KEY}"},
    json={"lat": -3.4653, "lng": -62.2159, "size": 10}
)
data = response.json()
print(f"Forest Score: {data['score']}/100")
print(f"Canopy Height: {data['canopy_height']}m (NASA GEDI)")
print(f"Deforestation Alert: {data['deforestation_alert']}")

const response = await fetch('https://www.canopysat.org/api/v1/analyze', {
